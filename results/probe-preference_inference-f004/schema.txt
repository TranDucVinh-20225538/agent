CREATE TABLE restaurants (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                cuisine TEXT,
                price_range TEXT,
                neighborhood TEXT,
                city TEXT,
                rating REAL,
                review_count INTEGER DEFAULT 0,
                has_outdoor INTEGER DEFAULT 0,
                has_tasting_menu INTEGER DEFAULT 0,
                is_vegetarian_friendly INTEGER DEFAULT 0,
                has_private_dining INTEGER DEFAULT 0,
                has_bar INTEGER DEFAULT 0,
                distance_miles REAL DEFAULT 0,
                popularity_score INTEGER DEFAULT 0,
                dietary_options TEXT DEFAULT '',
                description TEXT,
                address TEXT,
                hours TEXT,
                image_url TEXT,
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT ''
            );
CREATE TABLE sqlite_sequence(name,seq);
CREATE TABLE reviews (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                restaurant_id INTEGER,
                reservation_id INTEGER,
                user_email TEXT,
                user_name TEXT,
                rating INTEGER,
                food_rating INTEGER,
                service_rating INTEGER,
                ambiance_rating INTEGER,
                text TEXT,
                date TEXT,
                dined_on TEXT,
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT '',
                FOREIGN KEY (restaurant_id) REFERENCES restaurants(id),
                FOREIGN KEY (reservation_id) REFERENCES reservations(id)
            );
CREATE TABLE reservations (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_email TEXT NOT NULL,
                restaurant_id INTEGER,
                date TEXT,
                time TEXT,
                party_size INTEGER,
                special_requests TEXT,
                confirmation_code TEXT,
                status TEXT DEFAULT 'confirmed',
                slot_id INTEGER,
                cancellation_reason TEXT,
                cancelled_at TEXT,
                modified_at TEXT,
                honored_at TEXT,
                reward_points_awarded INTEGER DEFAULT 0,
                reward_awarded_at TEXT,
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT '',
                FOREIGN KEY (restaurant_id) REFERENCES restaurants(id),
                FOREIGN KEY (slot_id) REFERENCES slot_inventory(id)
            );
CREATE TABLE saved_lists (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_email TEXT NOT NULL,
                name TEXT,
                restaurant_ids TEXT,
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT ''
            );
CREATE TABLE restaurant_photos (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                restaurant_id INTEGER NOT NULL,
                image_url TEXT NOT NULL,
                alt_text TEXT,
                sort_order INTEGER DEFAULT 0,
                is_primary INTEGER DEFAULT 0,
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT '',
                FOREIGN KEY (restaurant_id) REFERENCES restaurants(id)
            );
CREATE TABLE slot_inventory (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                restaurant_id INTEGER NOT NULL,
                slot_date TEXT NOT NULL,
                slot_time TEXT NOT NULL,
                party_size INTEGER NOT NULL,
                capacity INTEGER NOT NULL,
                booked_count INTEGER DEFAULT 0,
                popularity_score INTEGER DEFAULT 0,
                UNIQUE (restaurant_id, slot_date, slot_time, party_size),
                FOREIGN KEY (restaurant_id) REFERENCES restaurants(id)
            );
CREATE TABLE waitlist_entries (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_email TEXT NOT NULL,
                restaurant_id INTEGER NOT NULL,
                reservation_id INTEGER UNIQUE,
                date TEXT NOT NULL,
                preferred_time TEXT NOT NULL,
                party_size INTEGER NOT NULL,
                position INTEGER DEFAULT 1,
                status TEXT DEFAULT 'active',
                notes TEXT,
                created_at TEXT,
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT '',
                FOREIGN KEY (restaurant_id) REFERENCES restaurants(id),
                FOREIGN KEY (reservation_id) REFERENCES reservations(id)
            );
CREATE TABLE user_rewards (
                user_email TEXT PRIMARY KEY,
                points_balance INTEGER DEFAULT 0,
                credits_balance_cents INTEGER DEFAULT 0,
                lifetime_points INTEGER DEFAULT 0,
                updated_at TEXT,
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT ''
            );
CREATE TABLE reward_transactions (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_email TEXT NOT NULL,
                reservation_id INTEGER,
                transaction_type TEXT NOT NULL,
                points_delta INTEGER DEFAULT 0,
                credit_delta_cents INTEGER DEFAULT 0,
                description TEXT,
                created_at TEXT,
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT '',
                FOREIGN KEY (reservation_id) REFERENCES reservations(id)
            );
CREATE TABLE menus (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                restaurant_id INTEGER NOT NULL,
                name TEXT NOT NULL,
                description TEXT,
                price REAL NOT NULL,
                category TEXT NOT NULL DEFAULT 'Entree',
                is_vegetarian INTEGER DEFAULT 0,
                is_gluten_free INTEGER DEFAULT 0,
                sort_order INTEGER DEFAULT 0,
                FOREIGN KEY (restaurant_id) REFERENCES restaurants(id)
            );
CREATE TABLE _seed_marker (key TEXT PRIMARY KEY, value TEXT NOT NULL);
CREATE INDEX idx_restaurants_rating ON restaurants(rating DESC);
CREATE INDEX idx_slots_lookup ON slot_inventory(restaurant_id, slot_date, party_size);
CREATE INDEX idx_reservations_user ON reservations(user_email, date);
CREATE INDEX idx_reviews_restaurant ON reviews(restaurant_id, date DESC);
CREATE INDEX idx_waitlist_restaurant ON waitlist_entries(restaurant_id, date, preferred_time);
CREATE INDEX idx_menus_restaurant ON menus(restaurant_id, category, sort_order);
CREATE UNIQUE INDEX idx_saved_lists_user_name ON saved_lists(user_email, name);
CREATE UNIQUE INDEX idx_reservations_active_uniq
    ON reservations(user_email, restaurant_id, date, time)
    WHERE status IN ('confirmed', 'waitlisted')
  ;
