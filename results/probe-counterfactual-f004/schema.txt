CREATE VIEW mailbox_email_entries AS
    SELECT
      emails.id,
      emails.user_email,
      mail_entry_state.folder AS folder,
      emails.from_name,
      emails.from_email,
      emails.to_email,
      emails.subject,
      emails.body,
      emails.body_html,
      emails.date,
      mail_entry_state.read AS read,
      mail_entry_state.starred AS starred,
      mail_entry_state.important AS important,
      emails.has_attachment,
      emails.attachment_name,
      emails.attachment_count,
      mail_entry_state.labels AS labels,
      emails.category,
      mail_entry_state.snooze_until AS snooze_until,
      mail_entry_state.snoozed_from_folder AS snoozed_from_folder,
      emails.subject_normalized,
      emails.thread_id,
      emails.in_reply_to,
      emails.canonical_id,
      emails.world_id,
      emails.vm_id,
      emails.actor_id
    FROM emails
    INNER JOIN mail_entry_state
      ON mail_entry_state.user_email = emails.user_email
     AND mail_entry_state.email_id = emails.id
