---SCHEMA---
CREATE TABLE loyalty (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_email TEXT NOT NULL,
                status TEXT NOT NULL DEFAULT 'Silver Medallion',
                miles INTEGER NOT NULL DEFAULT 0,
                miles_ytd INTEGER NOT NULL DEFAULT 0,
                medallion_qualifying_miles INTEGER NOT NULL DEFAULT 0,
                member_since TEXT NOT NULL,
                skymiles_number TEXT NOT NULL,
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT ''
            );
CREATE TABLE sqlite_sequence(name,seq);
CREATE TABLE flights (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_email TEXT NOT NULL,
                flight_number TEXT NOT NULL,
                origin TEXT NOT NULL,
                origin_city TEXT NOT NULL,
                destination TEXT NOT NULL,
                destination_city TEXT NOT NULL,
                departure_date TEXT NOT NULL,
                departure_time TEXT NOT NULL,
                arrival_date TEXT,
                arrival_time TEXT NOT NULL,
                status TEXT NOT NULL DEFAULT 'scheduled',
                seat TEXT,
                cabin_class TEXT NOT NULL DEFAULT 'Economy',
                gate TEXT,
                terminal TEXT,
                aircraft TEXT,
                duration TEXT,
                miles_earned INTEGER DEFAULT 0,
                confirmation_code TEXT NOT NULL,
                checked_in INTEGER DEFAULT 0,
                operational_status TEXT DEFAULT 'On Time',
                delay_minutes INTEGER DEFAULT 0,
                status_message TEXT,
                estimated_departure_time TEXT,
                estimated_arrival_time TEXT,
                arrival_terminal TEXT,
                arrival_gate TEXT,
                boarding_zone TEXT,
                boarding_time TEXT,
                fare_paid INTEGER DEFAULT 0,
                last_modified_at TEXT,
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT ''
            , marketing_carrier TEXT, marketing_flight_number TEXT);
CREATE TABLE boarding_passes (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                flight_id INTEGER NOT NULL,
                user_email TEXT NOT NULL,
                passenger_name TEXT NOT NULL,
                barcode TEXT NOT NULL,
                qr_payload TEXT,
                boarding_group TEXT,
                boarding_position TEXT,
                tsa_precheck INTEGER DEFAULT 0,
                wallet_stub_label TEXT,
                issued_at TEXT NOT NULL,
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT '',
                FOREIGN KEY (flight_id) REFERENCES flights(id)
            );
CREATE TABLE passenger_profiles (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_email TEXT NOT NULL UNIQUE,
                full_name TEXT NOT NULL,
                date_of_birth TEXT NOT NULL,
                gender TEXT NOT NULL,
                tsa_precheck INTEGER NOT NULL DEFAULT 0,
                passport_number TEXT NOT NULL,
                updated_at TEXT NOT NULL,
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT ''
            );
CREATE TABLE flight_passengers (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                flight_id INTEGER NOT NULL UNIQUE,
                user_email TEXT NOT NULL,
                full_name TEXT NOT NULL,
                date_of_birth TEXT NOT NULL,
                gender TEXT NOT NULL,
                tsa_precheck INTEGER NOT NULL DEFAULT 0,
                passport_number TEXT NOT NULL,
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT '',
                FOREIGN KEY (flight_id) REFERENCES flights(id)
            );
CREATE TABLE flight_baggage (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                flight_id INTEGER NOT NULL UNIQUE,
                user_email TEXT NOT NULL,
                checked_bag_count INTEGER NOT NULL DEFAULT 0,
                first_bag_free INTEGER NOT NULL DEFAULT 0,
                total_price INTEGER NOT NULL DEFAULT 0,
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT '',
                FOREIGN KEY (flight_id) REFERENCES flights(id)
            );
CREATE TABLE flight_seats (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                flight_id INTEGER NOT NULL,
                seat_number TEXT NOT NULL,
                cabin TEXT NOT NULL,
                row_number INTEGER NOT NULL,
                seat_letter TEXT NOT NULL,
                status TEXT NOT NULL DEFAULT 'available',
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT '',
                FOREIGN KEY (flight_id) REFERENCES flights(id)
            );
CREATE TABLE flight_alternatives (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                flight_id INTEGER NOT NULL,
                alternative_key TEXT,
                flight_number TEXT NOT NULL,
                departure_date TEXT NOT NULL,
                departure_time TEXT NOT NULL,
                arrival_date TEXT,
                arrival_time TEXT NOT NULL,
                estimated_departure_time TEXT NOT NULL,
                estimated_arrival_time TEXT NOT NULL,
                duration TEXT NOT NULL,
                aircraft TEXT NOT NULL,
                gate TEXT,
                terminal TEXT,
                arrival_gate TEXT,
                arrival_terminal TEXT,
                operational_status TEXT NOT NULL,
                delay_minutes INTEGER NOT NULL DEFAULT 0,
                status_message TEXT,
                fare_difference INTEGER NOT NULL DEFAULT 0,
                seats_left INTEGER NOT NULL DEFAULT 0,
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT '',
                FOREIGN KEY (flight_id) REFERENCES flights(id)
            );
CREATE TABLE flight_upgrades (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                flight_id INTEGER NOT NULL,
                upgrade_key TEXT,
                cabin_class TEXT NOT NULL,
                price_difference INTEGER NOT NULL,
                seat_hint TEXT,
                perks TEXT,
                status TEXT NOT NULL DEFAULT 'available',
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT '',
                FOREIGN KEY (flight_id) REFERENCES flights(id)
            );
CREATE TABLE payment_methods (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_email TEXT NOT NULL,
                cardholder_name TEXT NOT NULL,
                card_brand TEXT NOT NULL,
                last4 TEXT NOT NULL,
                exp_month TEXT NOT NULL,
                exp_year TEXT NOT NULL,
                billing_zip TEXT NOT NULL,
                is_default INTEGER NOT NULL DEFAULT 0,
                created_at TEXT NOT NULL,
                updated_at TEXT NOT NULL,
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT ''
            );
CREATE TABLE travel_credits (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_email TEXT NOT NULL,
                amount INTEGER NOT NULL,
                balance_remaining INTEGER NOT NULL DEFAULT 0,
                created_at TEXT NOT NULL,
                expires_at TEXT,
                source_booking_id INTEGER,
                redeemed INTEGER NOT NULL DEFAULT 0,
                world_id TEXT DEFAULT '',
                actor_id TEXT DEFAULT '',
                vm_id TEXT DEFAULT ''
            );
CREATE TABLE _seed_marker (key TEXT PRIMARY KEY, value TEXT NOT NULL);
CREATE INDEX idx_loyalty_user ON loyalty(user_email);
CREATE INDEX idx_flights_user ON flights(user_email);
CREATE INDEX idx_flights_date ON flights(departure_date);
CREATE INDEX idx_boarding_flight ON boarding_passes(flight_id);
CREATE UNIQUE INDEX idx_boarding_flight_user ON boarding_passes(flight_id, user_email);
CREATE INDEX idx_passenger_profile_user ON passenger_profiles(user_email);
CREATE INDEX idx_flight_passengers_user ON flight_passengers(user_email);
CREATE INDEX idx_flight_baggage_user ON flight_baggage(user_email);
CREATE UNIQUE INDEX idx_flight_seats_unique ON flight_seats(flight_id, seat_number);
CREATE INDEX idx_flight_alternatives_flight ON flight_alternatives(flight_id);
CREATE UNIQUE INDEX idx_flight_alternatives_key
      ON flight_alternatives(alternative_key);
CREATE INDEX idx_flight_upgrades_flight ON flight_upgrades(flight_id);
CREATE UNIQUE INDEX idx_flight_upgrades_key
      ON flight_upgrades(upgrade_key);
CREATE INDEX idx_payment_methods_user ON payment_methods(user_email);
CREATE INDEX idx_travel_credits_user ON travel_credits(user_email);
